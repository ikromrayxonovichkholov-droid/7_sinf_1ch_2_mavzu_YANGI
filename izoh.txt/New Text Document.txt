<script>

/* Barcha videolarni olish */

const videolar = document.querySelectorAll("video");

const darslar = document.querySelectorAll(".dars");


/*
   Har bir videoga "ended" hodisasini beramiz.
*/

videolar.forEach(function(video, index) {

    video.addEventListener("ended", function() {

        /* Hozirgi darsni faol holatdan chiqarish */

        darslar[index].classList.remove("faol");


        /* "Video tugadi" yozuvini chiqarish */

        const yozuv =
            darslar[index].querySelector(".tugadi");

        if (yozuv) {
            yozuv.style.display = "block";
        }


        /*
           Keyingi video mavjud bo'lsa,
           unga o'tamiz.
        */

        if (index + 1 < videolar.length) {

            const keyingiVideo =
                videolar[index + 1];

            const keyingiDars =
                darslar[index + 1];


            /* Keyingi darsni faol qilamiz */

            keyingiDars.classList.add("faol");


            /*
               Ekranni keyingi darsga
               avtomatik olib boramiz.
            */

            keyingiDars.scrollIntoView({
                behavior: "smooth",
                block: "center"
            });


            /*
               Keyingi videoni avtomatik
               ijro qilish.
            */

            setTimeout(function() {

                keyingiVideo.play()
                    .catch(function() {

                        /*
                           Agar brauzer avtomatik
                           ijroga ruxsat bermasa,
                           foydalanuvchi Play tugmasini
                           bosadi.
                        */

                        console.log(
                            "Videoni Play tugmasi bilan boshlang."
                        );

                    });

            }, 1000);

        }

        else {

            /*
               Barcha videolar tugaganda
               xabar chiqaramiz.
            */

            alert(
                "🎉 Tabriklaymiz! Barcha video darslar tugadi."
            );

        }

    });

});

</script>
